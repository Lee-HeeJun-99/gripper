cmake_minimum_required(VERSION 3.22)
project(ReGripFilmRoutedTransmission LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.LinearAlgebra REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
find_package(Eigen3 REQUIRED)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Field.generated.cpp")
 message(FATAL_ERROR "Use guarded routed runner to generate and hash the isolated Field source")
endif()
add_library(ReGripFilmRoutedTransmission SHARED Field.generated.cpp Owner.cpp)
add_executable(test_film_routed_api ../../tests/test_film_routed_api.generated.cpp)
foreach(target ReGripFilmRoutedTransmission test_film_routed_api)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -Werror=multichar -ffp-contract=off)
 target_link_libraries(${target} PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.LinearAlgebra Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
