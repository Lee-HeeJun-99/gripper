cmake_minimum_required(VERSION 3.22)
project(ReGripStagePower LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripStagePower SHARED StagePowerObserver.cpp)
target_compile_features(ReGripStagePower PRIVATE cxx_std_17)
target_compile_options(ReGripStagePower PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
target_link_libraries(ReGripStagePower PRIVATE Sofa.Core Sofa.Simulation.Core Eigen3::Eigen)
