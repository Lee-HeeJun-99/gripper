cmake_minimum_required(VERSION 3.22)
project(ReGripClosureDiagnostics LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.Component.SolidMechanics.FEM.Elastic REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../grasp/BeamEnergyProbe.cpp" base)
string(REPLACE "int main(int argc,char**argv)" "int unused_gradient_probe(int argc,char**argv)" base "${base}")
string(REPLACE "#include \"../integrated/Driver.cpp\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/../integrated/Driver.cpp\"" base "${base}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/BeamProbe.inc" CONTENT "${base}")
add_executable(closure_energy EnergyLoop.cpp)
add_executable(closure_memory MemoryReload.cpp)
foreach(target closure_energy closure_memory)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${target} PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.Component.SolidMechanics.FEM.Elastic dl)
endforeach()
