cmake_minimum_required(VERSION 3.22)
project(ReGripStationaryGaussPilot LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.LinearAlgebra REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripStationaryGaussPilot SHARED GaussOwner.cpp ../film_lateral_tendon/Field.generated.cpp)
target_compile_features(ReGripStationaryGaussPilot PRIVATE cxx_std_17)
target_compile_options(ReGripStationaryGaussPilot PRIVATE -O2 -Wall -Wextra -Werror=multichar -ffp-contract=off)
target_link_libraries(ReGripStationaryGaussPilot PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.LinearAlgebra Sofa.Simulation.Core)
