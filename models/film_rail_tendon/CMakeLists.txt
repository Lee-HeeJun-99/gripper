cmake_minimum_required(VERSION 3.22)
project(ReGripFilmRailTendon LANGUAGES CXX)
option(REGRIP_FILM_REUSE_LIFT "Experimental reusable element lift workspace;requires measured qualification" OFF)
if(REGRIP_FILM_REUSE_LIFT)
 add_compile_definitions(REGRIP_FILM_REUSE_LIFT)
endif()
option(REGRIP_FILM_DIRECT_VISITOR "Experimental compile-time element dispatch;requires bitwise/runtime qualification" OFF)
if(REGRIP_FILM_DIRECT_VISITOR)
 add_compile_definitions(REGRIP_FILM_DIRECT_VISITOR)
endif()
option(REGRIP_FILM_RAIL_NATIVE "Opt-in native ForceField vectorization,without fast-math" OFF)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripFilmRailTendon SHARED FilmRailTendon.cpp RootMapping.cpp)
add_executable(test_film_rail_tendon ../../tests/test_film_rail_tendon.cpp)
add_executable(test_film_rail_tendon_api ../../tests/test_film_rail_tendon_api.cpp)
add_executable(test_film_rail_mappings ../../tests/test_film_rail_mappings.cpp)
add_executable(test_film_control_profile ../../tests/test_film_control_profile.cpp)
target_compile_features(test_film_control_profile PRIVATE cxx_std_17)
target_compile_options(test_film_control_profile PRIVATE -O2 -ffp-contract=off)
foreach(target ReGripFilmRailTendon test_film_rail_tendon test_film_rail_tendon_api test_film_rail_mappings)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
endforeach()
if(REGRIP_FILM_RAIL_NATIVE)
 set_source_files_properties(FilmRailTendon.cpp PROPERTIES COMPILE_OPTIONS "-O3;-march=native;-fno-fast-math;-ffp-contract=off")
 foreach(target test_film_rail_tendon test_film_rail_tendon_api)
  target_compile_options(${target} PRIVATE -O3 -march=native -fno-fast-math -ffp-contract=off)
 endforeach()
endif()
foreach(target ReGripFilmRailTendon test_film_rail_tendon_api test_film_rail_mappings)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer Eigen3::Eigen)
endforeach()
