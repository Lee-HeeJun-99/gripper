cmake_minimum_required(VERSION 3.22)
project(ReGripIntegrated LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
add_library(ReGripIntegrated SHARED Components.cpp)
target_compile_features(ReGripIntegrated PUBLIC cxx_std_17)
target_compile_options(ReGripIntegrated PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ReGripIntegrated PUBLIC Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core)
add_executable(test_integrated ../../tests/test_integrated.cpp)
target_compile_features(test_integrated PUBLIC cxx_std_17)
target_compile_options(test_integrated PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(test_integrated PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
enable_testing()
add_test(NAME integrated_kernels COMMAND test_integrated)
