cmake_minimum_required(VERSION 3.22)
project(CurvedContactInvariants LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
if(NOT DEFINED FROZEN_CURVE_BUILD)
 message(FATAL_ERROR "Explicit compiled-generation directory required")
endif()
add_executable(curved_invariants Invariants.cpp)
target_include_directories(curved_invariants PRIVATE "${FROZEN_CURVE_BUILD}" "${FROZEN_CURVE_BUILD}/material_template/birth_template/legacy" ../../models/integrated)
target_compile_features(curved_invariants PRIVATE cxx_std_17)
target_compile_options(curved_invariants PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(curved_invariants PRIVATE Sofa.Core Sofa.Simulation.Core Sofa.Component.StateContainer)
