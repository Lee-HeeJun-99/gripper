cmake_minimum_required(VERSION 3.22)
project(ReGripFilmNativeObserver LANGUAGES CXX)
option(REGRIP_FILM_COUPLED "Experimental fixed-sphere single-plane monolithic correction" OFF)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.Constraint.Lagrangian.Solver REQUIRED)
add_library(ReGripFilmNativeObserver SHARED Observer.cpp)
if(REGRIP_FILM_COUPLED)
 find_package(Eigen3 REQUIRED)
 find_package(Sofa.Component.Constraint.Lagrangian.Model REQUIRED)
 target_sources(ReGripFilmNativeObserver PRIVATE CoupledSphereCorrection.cpp CoupledNativeCorrection.cpp)
 target_compile_definitions(ReGripFilmNativeObserver PRIVATE REGRIP_FILM_COUPLED REGRIP_FILM_REUSE_LIFT)
 target_link_libraries(ReGripFilmNativeObserver PRIVATE Eigen3::Eigen Sofa.Component.Constraint.Lagrangian.Model)
 add_executable(test_film_coupled_contacts ../../tests/test_film_coupled_contacts.cpp)
 target_compile_features(test_film_coupled_contacts PRIVATE cxx_std_17)
 target_compile_options(test_film_coupled_contacts PRIVATE -O2 -ffp-contract=off)
 target_link_libraries(test_film_coupled_contacts PRIVATE Eigen3::Eigen)
 add_executable(test_film_sphere_step ../../tests/test_film_sphere_step.cpp)
 target_compile_features(test_film_sphere_step PRIVATE cxx_std_17)
 target_compile_options(test_film_sphere_step PRIVATE -O2 -ffp-contract=off)
 target_link_libraries(test_film_sphere_step PRIVATE Eigen3::Eigen)
 add_executable(test_film_sphere_bdf ../../tests/test_film_sphere_bdf.cpp)
 target_compile_features(test_film_sphere_bdf PRIVATE cxx_std_17)
 target_compile_options(test_film_sphere_bdf PRIVATE -O2 -ffp-contract=off)
 target_link_libraries(test_film_sphere_bdf PRIVATE Eigen3::Eigen)
 add_executable(test_film_coupled_plane ../../tests/test_film_coupled_plane.cpp)
 target_compile_features(test_film_coupled_plane PRIVATE cxx_std_17)
 target_compile_options(test_film_coupled_plane PRIVATE -O2 -ffp-contract=off)
 target_link_libraries(test_film_coupled_plane PRIVATE Eigen3::Eigen)
 # Kernel helper definitions must not interpose on archived assembly DSOs.
 set_source_files_properties(CoupledSphereCorrection.cpp CoupledNativeCorrection.cpp PROPERTIES COMPILE_OPTIONS "-O3;-march=native;-fno-fast-math;-ffp-contract=off;-fvisibility=hidden;-fvisibility-inlines-hidden")
endif()
target_compile_features(ReGripFilmNativeObserver PRIVATE cxx_std_17)
target_compile_options(ReGripFilmNativeObserver PRIVATE -O2 -Wall -Wextra)
target_link_libraries(ReGripFilmNativeObserver PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.Constraint.Lagrangian.Solver)
