if(DEFINED FROZEN_FRAME_BUILD)
 file(READ "${FROZEN_FRAME_BUILD}/Integrated.inc" actualFrameCommon)
 set(classProof "{")
 foreach(pair IN ITEMS "EnergyField;LayerNetwork" "RoutedCable;AnalyticContact")
  list(GET pair 0 className)
  list(GET pair 1 nextName)
  string(FIND "${common}" "class ${className}" a)
  string(FIND "${common}" "class ${nextName}" b)
  math(EXPR length "${b}-${a}")
  string(SUBSTRING "${common}" ${a} ${length} expectedClass)
  string(FIND "${actualFrameCommon}" "class ${className}" c)
  string(FIND "${actualFrameCommon}" "class ${nextName}" d)
  math(EXPR length "${d}-${c}")
  string(SUBSTRING "${actualFrameCommon}" ${c} ${length} actualClass)
  if(NOT expectedClass STREQUAL actualClass)
   message(FATAL_ERROR "Frozen ${className} definitions differ;do not alias classes across ABI/model versions")
  endif()
  string(SHA256 classHash "${actualClass}")
  string(APPEND classProof "\"${className}\":\"${classHash}\",")
 endforeach()
 string(APPEND classProof "\"identical_frozen_class_definitions\":true}")
 file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ClassIdentity.json" CONTENT "${classProof}")
