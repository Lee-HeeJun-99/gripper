cmake_minimum_required(VERSION 3.22)
project(ReGripGeneralVectorContact LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
add_subdirectory(../curved_contact "${CMAKE_CURRENT_BINARY_DIR}/curved_template" EXCLUDE_FROM_ALL)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/GeneralGenerated.inc")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(GENERAL_HISTORY_GAUSS "Consistent fixed-material Gauss force/commit contrast" OFF)
option(GENERAL_HISTORY_ADAPTIVE "Experimental embedded material-rule estimator" OFF)
set(history_mode midpoint)
if(GENERAL_HISTORY_GAUSS)
 set(history_mode gauss)
endif()
if(GENERAL_HISTORY_ADAPTIVE)
 set(history_mode adaptive)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_curved_contact.cpp" test_body)
string(REPLACE "#include \"../models/curved_contact/CurvedContact.cpp\"" "" test_body "${test_body}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/GeneralTestBody.inc" "${test_body}")
add_custom_command(OUTPUT "${generated}"
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${CMAKE_CURRENT_BINARY_DIR}/curved_template/CurvedGenerated.inc" "${generated}" "${history_mode}"
 DEPENDS generate.py ReGripCurvedContact VERBATIM)
add_library(ReGripGeneralVectorContact SHARED GeneralCurvedContact.cpp "${generated}")
add_executable(test_general_curved ../../tests/test_general_curved.cpp "${generated}")
add_executable(test_general_wrench ../../tests/test_general_wrench.cpp "${generated}")
add_executable(test_general_history_owner ../../tests/test_general_history_owner.cpp)
target_compile_features(test_general_history_owner PRIVATE cxx_std_17)
target_compile_options(test_general_history_owner PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
foreach(target ReGripGeneralVectorContact test_general_curved test_general_wrench)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/curved_template/material_template/birth_template/legacy" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
