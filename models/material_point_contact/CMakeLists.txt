cmake_minimum_required(VERSION 3.22)
project(ReGripMaterialPointContact LANGUAGES CXX)
add_subdirectory(../contact_birth "${CMAKE_CURRENT_BINARY_DIR}/birth_template" EXCLUDE_FROM_ALL)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Sofa.Component.StateContainer REQUIRED)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/MaterialPointGenerated.inc")
add_custom_command(OUTPUT "${generated}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${CMAKE_CURRENT_BINARY_DIR}/birth_template/BirthGenerated.inc" "${generated}" DEPENDS generate.py "${CMAKE_CURRENT_BINARY_DIR}/birth_template/BirthGenerated.inc" VERBATIM)
add_library(ReGripMaterialPointContact SHARED MaterialPointContact.cpp "${generated}")
add_executable(test_material_point ../../tests/test_material_point.cpp "${generated}")
foreach(target ReGripMaterialPointContact test_material_point)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/birth_template/legacy" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
