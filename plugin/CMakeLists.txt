cmake_minimum_required(VERSION 3.22)
project(SofaTangentialV110 VERSION 1.1.1 LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
add_library(SofaTangentialV110 SHARED src/TangentialRigidForceField.cpp
    ../handover/20261114/SOFA_V110_Native_Tangent/src/initPlugin.cpp)
target_compile_features(SofaTangentialV110 PRIVATE cxx_std_17)
target_link_libraries(SofaTangentialV110 PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper)
set_target_properties(SofaTangentialV110 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS SofaTangentialV110 LIBRARY DESTINATION lib)
include(CTest)
if(BUILD_TESTING)
  find_package(Sofa.Component.StateContainer REQUIRED)
  add_executable(test_forcefield ../tests/test_forcefield.cpp)
  target_compile_features(test_forcefield PRIVATE cxx_std_17)
  target_link_libraries(test_forcefield PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Component.StateContainer)
  add_test(NAME native_force_energy_matrix COMMAND test_forcefield)
  add_executable(probe_forcefield ../tests/probe_forcefield.cpp)
  target_compile_features(probe_forcefield PRIVATE cxx_std_17)
  target_link_libraries(probe_forcefield PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Component.StateContainer)
endif()
