cmake_minimum_required(VERSION 3.22)
project(ReGripFilmShell LANGUAGES CXX)
option(REGRIP_FILM_NATIVE_OPT "Opt-in native vectorization of film ForceField only;not fast-math" OFF)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
find_package(Sofa.Component.ODESolver.Backward REQUIRED)
find_package(Sofa.Component.LinearSystem REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripFilmShell SHARED FilmShell.cpp ReGripBDFSolver.cpp FilmSlopeMapping.cpp FilmCubicSlopeMapping.cpp ReGripMatrixLinearSystem.cpp FilmRampLoad.cpp)
add_executable(test_film_shell ../../tests/test_film_shell.cpp)
add_executable(test_film_native_be ../../tests/test_film_native_be.cpp ReGripBDFSolver.cpp)
target_compile_features(test_film_native_be PRIVATE cxx_std_17)
target_include_directories(test_film_native_be PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../.runtime/sofa-source/Sofa/Component/ODESolver/Backward/src")
target_link_libraries(test_film_native_be PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.ODESolver.Backward Sofa.Component.StateContainer Eigen3::Eigen)
add_executable(test_film_shell_api ../../tests/test_film_shell_api.cpp)
target_sources(test_film_shell_api PRIVATE ReGripBDFSolver.cpp FilmSlopeMapping.cpp FilmCubicSlopeMapping.cpp ReGripMatrixLinearSystem.cpp FilmRampLoad.cpp)
target_link_libraries(test_film_shell_api PRIVATE Sofa.Component.ODESolver.Backward)
target_link_libraries(test_film_shell_api PRIVATE Sofa.Component.LinearSystem)
target_link_libraries(test_film_shell_api PRIVATE Eigen3::Eigen)
# The matched SDK omits NonLinearFunction.h from its install tree. Read the
# public header from the matching source; do not modify installed SDK files.
set(REGRIP_MATCHING_BACKWARD_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.runtime/sofa-source/Sofa/Component/ODESolver/Backward/src")
target_include_directories(ReGripFilmShell PRIVATE "${REGRIP_MATCHING_BACKWARD_SOURCE}")
target_include_directories(test_film_shell_api PRIVATE "${REGRIP_MATCHING_BACKWARD_SOURCE}")
add_executable(test_film_plane ../../tests/test_film_plane.cpp)
add_executable(test_film_compact ../../tests/test_film_compact.cpp)
add_executable(test_film_bdf_coefficients ../../tests/test_film_bdf_coefficients.cpp)
target_compile_features(test_film_bdf_coefficients PRIVATE cxx_std_17)
target_compile_features(test_film_compact PRIVATE cxx_std_17)
target_compile_options(test_film_compact PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
add_executable(test_film_slope_mapping ../../tests/test_film_slope_mapping.cpp)
add_executable(test_film_cubic_slope ../../tests/test_film_cubic_slope.cpp)
target_compile_features(test_film_cubic_slope PRIVATE cxx_std_17)
target_compile_options(test_film_cubic_slope PRIVATE -O2 -ffp-contract=off)
target_link_libraries(test_film_cubic_slope PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer Eigen3::Eigen)
target_link_libraries(test_film_slope_mapping PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer Eigen3::Eigen)
add_executable(film_static ../../diagnostics/film_shell/Static.cpp)
add_executable(film_implicit_reference ../../diagnostics/film_shell/ImplicitReference.cpp)
add_executable(film_mapped_tangent ../../diagnostics/film_shell/MappedTangent.cpp)
add_executable(film_trapezoid_reference ../../diagnostics/film_shell/TrapezoidReference.cpp)
target_compile_features(film_trapezoid_reference PRIVATE cxx_std_17)
target_compile_options(film_trapezoid_reference PRIVATE -O2 -ffp-contract=off)
target_link_libraries(film_trapezoid_reference PRIVATE Eigen3::Eigen)
target_compile_features(film_mapped_tangent PRIVATE cxx_std_17)
target_compile_options(film_mapped_tangent PRIVATE -O2 -ffp-contract=off)
target_link_libraries(film_mapped_tangent PRIVATE Eigen3::Eigen)
target_link_libraries(film_static PRIVATE Eigen3::Eigen)
target_link_libraries(film_implicit_reference PRIVATE Eigen3::Eigen)
foreach(target ReGripFilmShell test_film_shell film_static test_film_shell_api test_film_plane test_film_slope_mapping film_implicit_reference)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
endforeach()
target_link_libraries(ReGripFilmShell PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer)
target_link_libraries(ReGripFilmShell PRIVATE Sofa.Component.ODESolver.Backward)
target_link_libraries(ReGripFilmShell PRIVATE Sofa.Component.LinearSystem)
target_link_libraries(ReGripFilmShell PRIVATE Eigen3::Eigen)
target_link_libraries(test_film_shell_api PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer)
if(REGRIP_FILM_NATIVE_OPT)
 # Keep ODE/matrix assembly translation units unchanged. No reassociation or
 # fused contractions: independent AD entries can be vectorized safely.
 set_source_files_properties(FilmShell.cpp PROPERTIES COMPILE_OPTIONS "-O3;-march=native;-fno-fast-math;-ffp-contract=off")
 foreach(target test_film_shell test_film_shell_api test_film_compact test_film_plane)
  target_compile_options(${target} PRIVATE -O3 -march=native -fno-fast-math -ffp-contract=off)
 endforeach()
endif()
