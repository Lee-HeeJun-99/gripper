cmake_minimum_required(VERSION 3.22)
project(ReGripFiniteRotation LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
add_library(ReGripFiniteRotation SHARED FiniteRotationShear.cpp)
target_compile_features(ReGripFiniteRotation PUBLIC cxx_std_17)
target_compile_options(ReGripFiniteRotation PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ReGripFiniteRotation PUBLIC Sofa.Core Sofa.DefaultType Sofa.Helper)
add_executable(test_finite_rotation ../../tests/test_finite_rotation.cpp)
target_compile_features(test_finite_rotation PUBLIC cxx_std_17)
target_compile_options(test_finite_rotation PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(test_finite_rotation PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Component.StateContainer)
enable_testing()
add_test(NAME finite_rotation_derivatives COMMAND test_finite_rotation)
