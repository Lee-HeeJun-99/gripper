cmake_minimum_required(VERSION 3.22)
project(ReGripFilmValue LANGUAGES CXX)
# Mechanically reuse original formulas in an ISOLATED namespace. All generated
# bytes/hashes are recorded. Do not alter original AD/Film/Assembly sources.
set(value_root "${CMAKE_CURRENT_BINARY_DIR}/value_models")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated_value_manifest.tsv" "sha256\tgenerated_path\n")
foreach(directory film_shell film_rail_tendon)
  file(MAKE_DIRECTORY "${value_root}/${directory}")
  file(GLOB headers "${CMAKE_CURRENT_SOURCE_DIR}/../${directory}/*.h")
  foreach(input IN LISTS headers)
    get_filename_component(name "${input}" NAME)
    file(READ "${input}" content)
    string(REPLACE "regrip_film_rail" "regrip_value_film_rail" content "${content}")
    string(REPLACE "regrip_film" "regrip_value_film" content "${content}")
    file(WRITE "${value_root}/${directory}/${name}" "${content}")
  endforeach()
endforeach()
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/ValueAD.h.in" "${value_root}/film_shell/AD.h" COPYONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../film_shell/MembraneAD.h" membrane)
string(FIND "${membrane}" "inline AD triangle(" triangle_start)
if(triangle_start LESS 0)
  message(FATAL_ERROR "original compact triangle formula delimiter changed")
endif()
string(SUBSTRING "${membrane}" ${triangle_start} -1 triangle_formula)
file(WRITE "${value_root}/film_shell/MembraneAD.h" "#pragma once\n#include \"AD.h\"\nnamespace regrip_value_film::compact {\nusing AD9=AD;\ninline AD expand(const AD&a){return a;}\n${triangle_formula}")
file(GLOB_RECURSE generated "${value_root}/*.h")
foreach(path IN LISTS generated)
  file(SHA256 "${path}" hash)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/generated_value_manifest.tsv" "${hash}\t${path}\n")
endforeach()
add_library(ReGripFilmValue STATIC ValueEnergy.cpp)
set_target_properties(ReGripFilmValue PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(ReGripFilmValue PRIVATE cxx_std_17)
target_compile_options(ReGripFilmValue PRIVATE -O2 -Wall -Wextra -ffp-contract=off -fvisibility=hidden -fvisibility-inlines-hidden)
target_include_directories(ReGripFilmValue PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
