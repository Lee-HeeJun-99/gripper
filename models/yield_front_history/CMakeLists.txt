cmake_minimum_required(VERSION 3.22)
project(ReGripVectorPatchReference LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
add_library(ReGripVectorPatchReference SHARED VectorPatchBenchmark.cpp)
add_executable(test_vector_patch ../../tests/test_vector_patch.cpp)
foreach(target ReGripVectorPatchReference test_vector_patch)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -O2 -ffp-contract=off -Wall -Wextra -Wpedantic -Werror)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
