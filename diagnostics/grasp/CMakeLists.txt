cmake_minimum_required(VERSION 3.22)
project(ReGripGraspDiagnostics LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
add_executable(memory_lifetime ../stabilization/MemoryLifetime.cpp)
add_executable(graph_probe ../integrated/Driver.cpp)
foreach(target memory_lifetime graph_probe)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${target} PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph dl)
endforeach()
option(REGRIP_BEAM_PROBE "Build independent BeamFEM energy differentiation probe" ON)
if(REGRIP_BEAM_PROBE)
 find_package(Sofa.Component.SolidMechanics.FEM.Elastic REQUIRED)
 add_executable(beam_energy_probe BeamEnergyProbe.cpp)
 target_compile_features(beam_energy_probe PRIVATE cxx_std_17)
 target_compile_options(beam_energy_probe PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(beam_energy_probe PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.Component.SolidMechanics.FEM.Elastic dl)
endif()
