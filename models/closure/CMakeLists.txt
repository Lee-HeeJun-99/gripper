cmake_minimum_required(VERSION 3.22)
project(ReGripClosure LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../integrated/Components.cpp" common)
string(REPLACE "void registerObjects(sofa::core::ObjectFactory*f)" "void registerIntegratedObjects(sofa::core::ObjectFactory*f)" common "${common}")
string(REPLACE "const char*getModuleName()" "const char*getIntegratedModuleName()" common "${common}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Integrated.inc" CONTENT "${common}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../grasp/StickSlip.cpp" stick)
string(REPLACE "#include \"StickSlip.h\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/../grasp/StickSlip.h\"" stick "${stick}")
string(REPLACE "void registerObjects(sofa::core::ObjectFactory*f)" "void registerStickObjects(sofa::core::ObjectFactory*f)" stick "${stick}")
string(REPLACE "const char*getModuleName()" "const char*getStickModuleName()" stick "${stick}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/StickBase.inc" CONTENT "${stick}")
add_library(ReGripClosure SHARED AdaptiveContact.cpp)
add_executable(test_closure ../../tests/test_closure.cpp)
foreach(target ReGripClosure test_closure)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
