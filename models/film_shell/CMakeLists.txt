cmake_minimum_required(VERSION 3.22)
project(ReGripFilmShell LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
find_package(Sofa.Component.ODESolver.Backward REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripFilmShell SHARED FilmShell.cpp ReGripBDFSolver.cpp)
add_executable(test_film_shell ../../tests/test_film_shell.cpp)
add_executable(test_film_shell_api ../../tests/test_film_shell_api.cpp)
target_sources(test_film_shell_api PRIVATE ReGripBDFSolver.cpp)
target_link_libraries(test_film_shell_api PRIVATE Sofa.Component.ODESolver.Backward)
# The matched SDK omits NonLinearFunction.h from its install tree. Read the
# public header from the matching source; do not modify installed SDK files.
set(REGRIP_MATCHING_BACKWARD_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.runtime/sofa-source/Sofa/Component/ODESolver/Backward/src")
target_include_directories(ReGripFilmShell PRIVATE "${REGRIP_MATCHING_BACKWARD_SOURCE}")
target_include_directories(test_film_shell_api PRIVATE "${REGRIP_MATCHING_BACKWARD_SOURCE}")
add_executable(test_film_plane ../../tests/test_film_plane.cpp)
add_executable(film_static ../../diagnostics/film_shell/Static.cpp)
add_executable(film_implicit_reference ../../diagnostics/film_shell/ImplicitReference.cpp)
target_link_libraries(film_static PRIVATE Eigen3::Eigen)
target_link_libraries(film_implicit_reference PRIVATE Eigen3::Eigen)
foreach(target ReGripFilmShell test_film_shell film_static test_film_shell_api test_film_plane film_implicit_reference)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
endforeach()
target_link_libraries(ReGripFilmShell PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer)
target_link_libraries(ReGripFilmShell PRIVATE Sofa.Component.ODESolver.Backward)
target_link_libraries(test_film_shell_api PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer)
