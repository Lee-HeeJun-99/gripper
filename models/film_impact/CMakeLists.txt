cmake_minimum_required(VERSION 3.22)
project(ReGripImpactReference LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripImpactReference SHARED PlaneImpactOwner.cpp FilmImpulseOwner.cpp)
target_compile_features(ReGripImpactReference PRIVATE cxx_std_17)
target_compile_options(ReGripImpactReference PRIVATE -O2 -Wall -Wextra -ffp-contract=off -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_libraries(ReGripImpactReference PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.Simulation.Core)
