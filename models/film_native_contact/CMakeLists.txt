cmake_minimum_required(VERSION 3.22)
project(ReGripFilmNativeObserver LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.Constraint.Lagrangian.Solver REQUIRED)
add_library(ReGripFilmNativeObserver SHARED Observer.cpp)
target_compile_features(ReGripFilmNativeObserver PRIVATE cxx_std_17)
target_compile_options(ReGripFilmNativeObserver PRIVATE -O2 -Wall -Wextra)
target_link_libraries(ReGripFilmNativeObserver PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.Constraint.Lagrangian.Solver)
