cmake_minimum_required(VERSION 3.22)
project(ReGripRigidImpactReference LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.LinearAlgebra REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.ODESolver.Backward REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripRigidImpactReference SHARED RigidPlane.cpp FilmRigidImpulse.cpp NativeFreeBodyProbe.cpp ReGripRigidBE.cpp)
target_include_directories(ReGripRigidImpactReference PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sdk_public")
target_compile_features(ReGripRigidImpactReference PRIVATE cxx_std_17)
target_compile_options(ReGripRigidImpactReference PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
target_link_libraries(ReGripRigidImpactReference PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.LinearAlgebra Sofa.Simulation.Core Sofa.Component.ODESolver.Backward)
