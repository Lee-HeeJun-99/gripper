cmake_minimum_required(VERSION 3.22)
project(ReGripContactReturnAudit LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
if(NOT DEFINED FROZEN_CURVE_BUILD)
 message(FATAL_ERROR "Frozen compiled project identity required")
endif()
file(READ "${FROZEN_CURVE_BUILD}/material_template/birth_template/legacy/Integrated.inc" common)
string(FIND "${common}" "extern \"C\"" cut)
string(SUBSTRING "${common}" 0 ${cut} common)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/IntegratedClassOnly.inc" CONTENT "${common}")
file(READ "${FROZEN_CURVE_BUILD}/CurvedGenerated.inc" curve)
string(FIND "${curve}" "extern \"C\"" cut)
string(SUBSTRING "${curve}" 0 ${cut} curve)
string(REPLACE "#include \"StickBase.inc\"" "#include \"IntegratedClassOnly.inc\"" curve "${curve}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/CurveClassOnly.inc" CONTENT "${curve}")
add_library(ContactReturnBridge SHARED Bridge.cpp)
target_include_directories(ContactReturnBridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../../models/integrated)
target_compile_features(ContactReturnBridge PRIVATE cxx_std_17)
target_compile_options(ContactReturnBridge PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_libraries(ContactReturnBridge PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
add_executable(contact_return_journal Driver.cpp)
target_compile_features(contact_return_journal PRIVATE cxx_std_17)
target_compile_options(contact_return_journal PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(contact_return_journal PRIVATE ContactReturnBridge Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
