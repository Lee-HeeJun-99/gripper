cmake_minimum_required(VERSION 3.22)
project(ReGripDampingMatrixAudit LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
add_executable(damping_matrix_audit Driver.cpp)
target_compile_features(damping_matrix_audit PRIVATE cxx_std_17)
target_compile_options(damping_matrix_audit PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(damping_matrix_audit PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
