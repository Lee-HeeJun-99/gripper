cmake_minimum_required(VERSION 3.22)
project(ReGripThreeFilmResume LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.LinearAlgebra REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripThreeFilmResume SHARED ThreeFilmResume.cpp)
target_compile_features(ReGripThreeFilmResume PRIVATE cxx_std_17)
target_compile_options(ReGripThreeFilmResume PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
target_link_libraries(ReGripThreeFilmResume PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.LinearAlgebra Sofa.Simulation.Core)
