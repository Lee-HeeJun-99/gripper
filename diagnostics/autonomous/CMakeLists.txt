cmake_minimum_required(VERSION 3.22)
project(ReGripPrescribedHistory LANGUAGES CXX)
add_subdirectory(../../models/material_point_contact "${CMAKE_CURRENT_BINARY_DIR}/component" EXCLUDE_FROM_ALL)
find_package(Sofa.Component.StateContainer REQUIRED)
foreach(name prescribed_old prescribed_new)
 add_executable(${name} Prescribed.cpp)
 add_dependencies(${name} ReGripMaterialPointContact)
 target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/component" "${CMAKE_CURRENT_BINARY_DIR}/component/birth_template" "${CMAKE_CURRENT_BINARY_DIR}/component/birth_template/legacy" "${CMAKE_CURRENT_SOURCE_DIR}/../../models/integrated")
 target_compile_features(${name} PRIVATE cxx_std_17)
 target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${name} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
target_compile_definitions(prescribed_old PRIVATE REGRIP_OLD_HISTORY)
