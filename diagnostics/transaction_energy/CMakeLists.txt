cmake_minimum_required(VERSION 3.22)
project(ReGripTransactionJournal LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
add_executable(transaction_journal Driver.cpp)
target_compile_features(transaction_journal PRIVATE cxx_std_17)
target_compile_options(transaction_journal PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(transaction_journal PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
option(REGRIP_NORMAL_AUDIT "Read-only normal force/Jacobian decomposition" OFF)
if(REGRIP_NORMAL_AUDIT)
 add_subdirectory(../contact_return_energy "${CMAKE_CURRENT_BINARY_DIR}/normal_bridge")
 target_compile_definitions(transaction_journal PRIVATE REGRIP_NORMAL_AUDIT=1)
 target_link_libraries(transaction_journal PRIVATE ContactReturnBridge)
 if(TARGET FrameReturnBridge)
  target_compile_definitions(transaction_journal PRIVATE REGRIP_FRAME_RETURN_AUDIT=1)
  target_link_libraries(transaction_journal PRIVATE FrameReturnBridge)
 endif()
endif()
option(REGRIP_VALUE_ENERGY "Use independently verified value-only contact energy" OFF)
if(REGRIP_VALUE_ENERGY)
 if(NOT REGRIP_NORMAL_AUDIT)
  message(FATAL_ERROR "Value-only energy requires the validated public project bridge")
 endif()
 target_compile_definitions(transaction_journal PRIVATE REGRIP_VALUE_ENERGY=1)
endif()
