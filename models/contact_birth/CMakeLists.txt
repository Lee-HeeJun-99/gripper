cmake_minimum_required(VERSION 3.22)
project(ReGripBirthContact LANGUAGES CXX)
add_subdirectory(../contact_history "${CMAKE_CURRENT_BINARY_DIR}/legacy" EXCLUDE_FROM_ALL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../contact_history/ContactFriction.cpp" source)
string(REPLACE "MaterialHistoryContact" "BirthAwareContact" source "${source}")
string(REPLACE "ReGripMaterialHistory" "ReGripBirthContact" source "${source}")
string(REPLACE "#include \"../closure/Quadrature.h\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/../closure/Quadrature.h\"" source "${source}")
string(REPLACE "#include \"MaterialHistory.h\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/BirthLaw.h\"" source "${source}")
string(FIND "${source}" " regrip_history::History old(" begin)
string(FIND "${source}" " regrip_closure::Interval interval(" end)
if(begin LESS 0 OR end LESS 0)
 message(FATAL_ERROR "frozen v3 template changed")
endif()
string(SUBSTRING "${source}" 0 ${begin} head)
string(SUBSTRING "${source}" ${end} -1 tail)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/HistoryCommit.inc" history)
set(source "${head}${history}${tail}")
string(REPLACE "auto po=pose(x[obj]);auto cells" "auto cells" source "${source}")
string(REPLACE "validateInput(x);Snapshot out" "validateInput(x);checkTemporal(x);Snapshot out" source "${source}")
string(REPLACE "validateInput(x);std::vector<std::vector" "validateInput(x);checkTemporal(x);std::vector<std::vector" source "${source}")
string(REGEX REPLACE "auto r=regrip_history::evaluateContact\\(pp,po,tangential[^;]+;" "auto r=query(x,edge,s,tangential);" source "${source}")
string(REGEX REPLACE "auto r=regrip_history::value\\(point\\(x,e,s\\)[^;]+;" "auto r=commitValue(x,e,s);" source "${source}")
string(REPLACE "try{const auto&x=mstate" "try{double eventTime=getContext()->getTime();if(eventTime==lastEventTime)return;lastEventTime=eventTime;const auto&x=mstate" source "${source}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/BirthGenerated.inc" CONTENT "${source}")
find_package(Sofa.Component.StateContainer REQUIRED)
add_library(ReGripBirthContact SHARED BirthAwareContact.cpp)
add_executable(test_birth ../../tests/test_birth.cpp)
foreach(target ReGripBirthContact test_birth)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/legacy" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
