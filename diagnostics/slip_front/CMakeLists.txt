cmake_minimum_required(VERSION 3.22)
project(ReGripSlipFrontProbe LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
if(NOT DEFINED FROZEN_AUDIT_BUILD)
 message(FATAL_ERROR "Existing frozen CurveClassOnly/IntegratedClassOnly required")
endif()
add_executable(slip_front_probe Driver.cpp)
target_compile_features(slip_front_probe PRIVATE cxx_std_17)
target_include_directories(slip_front_probe PRIVATE "${FROZEN_AUDIT_BUILD}" ../../models/integrated)
target_compile_options(slip_front_probe PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden -fvisibility-inlines-hidden -march=native -ffp-contract=off)
target_link_libraries(slip_front_probe PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
