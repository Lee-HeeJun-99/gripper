cmake_minimum_required(VERSION 3.22)
project(ReGripFilmTransmission LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripFilmTransmission SHARED Owner.cpp)
target_compile_features(ReGripFilmTransmission PRIVATE cxx_std_17)
target_compile_options(ReGripFilmTransmission PRIVATE -O2 -Wall -Wextra -Werror=multichar -ffp-contract=off)
target_link_libraries(ReGripFilmTransmission PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.Simulation.Core)
