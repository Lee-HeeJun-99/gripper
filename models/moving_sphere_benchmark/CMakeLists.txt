cmake_minimum_required(VERSION 3.22)
project(ReGripMovingSphereBenchmark LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
add_library(ReGripMovingSphereBenchmark SHARED ../general_vector_history/MovingSphereBenchmark.cpp)
add_executable(test_moving_sphere ../../tests/test_moving_sphere.cpp)
foreach(target ReGripMovingSphereBenchmark test_moving_sphere)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core Sofa.Component.StateContainer)
endforeach()
