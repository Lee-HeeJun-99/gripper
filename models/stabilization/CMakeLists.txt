cmake_minimum_required(VERSION 3.22)
project(ReGripStabilization LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../integrated/Components.cpp" integrated_source)
string(REPLACE "void registerObjects(sofa::core::ObjectFactory*f)" "void registerIntegratedObjects(sofa::core::ObjectFactory*f)" integrated_source "${integrated_source}")
string(REPLACE "const char*getModuleName()" "const char*getIntegratedModuleName()" integrated_source "${integrated_source}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Integrated.inc" CONTENT "${integrated_source}")
add_library(ReGripStabilization SHARED Audit.cpp)
target_include_directories(ReGripStabilization PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
target_compile_features(ReGripStabilization PUBLIC cxx_std_17)
target_compile_options(ReGripStabilization PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ReGripStabilization PUBLIC Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core)
add_library(ReGripDistributed SHARED Distributed.cpp)
target_include_directories(ReGripDistributed PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
target_compile_features(ReGripDistributed PUBLIC cxx_std_17)
target_compile_options(ReGripDistributed PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ReGripDistributed PUBLIC Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core)
find_package(Sofa.Component.StateContainer REQUIRED)
add_executable(test_stabilization_contact ../../tests/test_stabilization_contact.cpp)
target_include_directories(test_stabilization_contact PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
target_compile_features(test_stabilization_contact PRIVATE cxx_std_17)
target_compile_options(test_stabilization_contact PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(test_stabilization_contact PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
add_library(ReGripTelemetry SHARED Telemetry.cpp)
target_compile_features(ReGripTelemetry PRIVATE cxx_std_17)
target_compile_options(ReGripTelemetry PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ReGripTelemetry PRIVATE Sofa.Core Sofa.Simulation.Core)
add_library(ReGripCompensated SHARED Compensated.cpp)
target_compile_features(ReGripCompensated PRIVATE cxx_std_17)
target_compile_options(ReGripCompensated PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ReGripCompensated PRIVATE Sofa.Core Sofa.DefaultType Sofa.Simulation.Core)
add_executable(test_compensated ../../tests/test_compensated.cpp)
target_compile_features(test_compensated PRIVATE cxx_std_17)
