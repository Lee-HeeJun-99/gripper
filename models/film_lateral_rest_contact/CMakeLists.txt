cmake_minimum_required(VERSION 3.22)
project(ReGripStationaryImpact LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.LinearAlgebra REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripStationaryImpact SHARED StationaryImpact.cpp ../film_lateral_tendon/Field.generated.cpp)
target_compile_features(ReGripStationaryImpact PRIVATE cxx_std_17)
target_compile_options(ReGripStationaryImpact PRIVATE -O2 -Wall -Wextra -Werror=multichar -ffp-contract=off)
target_link_libraries(ReGripStationaryImpact PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.LinearAlgebra Sofa.Simulation.Core)
