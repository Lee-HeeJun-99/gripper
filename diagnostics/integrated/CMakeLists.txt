cmake_minimum_required(VERSION 3.22)
project(ReGripIntegratedDiagnostics LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
add_executable(integrated_driver Driver.cpp)
target_compile_features(integrated_driver PRIVATE cxx_std_17)
target_compile_options(integrated_driver PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(integrated_driver PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph dl)
add_library(AllocationProbe SHARED AllocationProbe.cpp)
target_compile_features(AllocationProbe PRIVATE cxx_std_17)
target_compile_options(AllocationProbe PRIVATE -Wall -Wextra -g -fno-omit-frame-pointer)
target_link_libraries(AllocationProbe PRIVATE dl pthread)
