cmake_minimum_required(VERSION 3.22)
project(ReGripTransactionJournal LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
add_executable(transaction_journal Driver.cpp)
target_compile_features(transaction_journal PRIVATE cxx_std_17)
target_compile_options(transaction_journal PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(transaction_journal PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
