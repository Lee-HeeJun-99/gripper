cmake_minimum_required(VERSION 3.22)
project(ReGripCoupledRadau LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.LinearAlgebra REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(ReGripCoupledRadau SHARED RadauOwner.cpp)
target_compile_features(ReGripCoupledRadau PRIVATE cxx_std_17)
target_compile_options(ReGripCoupledRadau PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
target_link_libraries(ReGripCoupledRadau PRIVATE Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.LinearAlgebra Sofa.Simulation.Core)
