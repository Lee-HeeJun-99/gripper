cmake_minimum_required(VERSION 3.22)
project(ReGripTransactionJournal LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
add_executable(transaction_journal Driver.cpp)
target_compile_features(transaction_journal PRIVATE cxx_std_17)
target_compile_options(transaction_journal PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(transaction_journal PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
option(REGRIP_NORMAL_AUDIT "Read-only normal force/Jacobian decomposition" OFF)
if(REGRIP_NORMAL_AUDIT)
 add_subdirectory(../contact_return_energy "${CMAKE_CURRENT_BINARY_DIR}/normal_bridge")
 target_compile_definitions(transaction_journal PRIVATE REGRIP_NORMAL_AUDIT=1)
 target_link_libraries(transaction_journal PRIVATE ContactReturnBridge)
 if(TARGET FrameReturnBridge)
  target_compile_definitions(transaction_journal PRIVATE REGRIP_FRAME_RETURN_AUDIT=1)
  target_link_libraries(transaction_journal PRIVATE FrameReturnBridge)
 endif()
endif()
option(REGRIP_CABLE_SNAPSHOT "Reuse exact tension-cable assembly;omit optional virtual wire gauge" OFF)
if(REGRIP_CABLE_SNAPSHOT)
 if(NOT TARGET FrameReturnBridge OR NOT REGRIP_FRAME_SNAPSHOT)
  message(FATAL_ERROR "Cable snapshot requires the validated frame snapshot/public bridge")
 endif()
 target_compile_definitions(transaction_journal PRIVATE REGRIP_CABLE_SNAPSHOT=1)
endif()
option(REGRIP_FRAME_SNAPSHOT "Reuse exact frame public assembly within one observer transaction" OFF)
if(REGRIP_FRAME_SNAPSHOT)
 if(NOT TARGET FrameReturnBridge)
  message(FATAL_ERROR "Frame snapshot requires the frozen-class public bridge")
 endif()
 target_compile_definitions(transaction_journal PRIVATE REGRIP_FRAME_SNAPSHOT=1)
endif()
option(REGRIP_GROUPED_CONTACT "Equivalent full-contact work grouping without extra normal Jacobian" OFF)
if(REGRIP_GROUPED_CONTACT)
 if(NOT REGRIP_NORMAL_AUDIT OR NOT REGRIP_VALUE_ENERGY)
  message(FATAL_ERROR "Grouped contact requires validated return/value energy bridges")
 endif()
 target_compile_definitions(transaction_journal PRIVATE REGRIP_GROUPED_CONTACT=1)
endif()
option(REGRIP_VALUE_ENERGY "Use independently verified value-only contact energy" OFF)
if(REGRIP_VALUE_ENERGY)
 if(NOT REGRIP_NORMAL_AUDIT)
  message(FATAL_ERROR "Value-only energy requires the validated public project bridge")
 endif()
 target_compile_definitions(transaction_journal PRIVATE REGRIP_VALUE_ENERGY=1)
endif()
option(REGRIP_REUSE_RETURN_ENERGY "Reuse identical return snapshot/unchanged post-commit energy with exact sampled checks" OFF)
if(REGRIP_REUSE_RETURN_ENERGY)
 if(NOT REGRIP_GROUPED_CONTACT)
  message(FATAL_ERROR "Energy reuse requires validated grouped/return observer")
 endif()
 target_compile_definitions(transaction_journal PRIVATE REGRIP_REUSE_RETURN_ENERGY=1)
endif()
