cmake_minimum_required(VERSION 3.22)
project(ReGripPhysicsCheckpoint LANGUAGES CXX)
find_package(Sofa.SimpleApi REQUIRED)
find_package(Sofa.Simulation.Graph REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
foreach(var FROZEN_CURVE_BUILD FROZEN_FRAME_BUILD)
 if(NOT DEFINED ${var})
  message(FATAL_ERROR "Frozen project-class generation required")
 endif()
endforeach()
file(READ "${FROZEN_CURVE_BUILD}/material_template/birth_template/legacy/Integrated.inc" common)
string(FIND "${common}" "extern \"C\"" cut)
string(SUBSTRING "${common}" 0 ${cut} common)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/IntegratedClassOnly.inc" CONTENT "${common}")
file(READ "${FROZEN_CURVE_BUILD}/CurvedGenerated.inc" curve)
string(FIND "${curve}" "extern \"C\"" cut)
string(SUBSTRING "${curve}" 0 ${cut} curve)
string(REPLACE "#include \"StickBase.inc\"" "#include \"IntegratedClassOnly.inc\"" curve "${curve}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/CurveClassOnly.inc" CONTENT "${curve}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../models/closure_frame/FrameLayer.cpp" frame)
string(FIND "${frame}" "extern \"C\"" cut)
string(SUBSTRING "${frame}" 0 ${cut} frame)
string(REPLACE "#include \"AdaptiveBase.inc\"" "#include \"IntegratedClassOnly.inc\"" frame "${frame}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/FrameClassOnly.inc" CONTENT "${frame}")
add_library(CurveCheckpointBridge SHARED CurveBridge.cpp)
add_library(FrameCheckpointBridge SHARED FrameBridge.cpp)
foreach(target CurveCheckpointBridge FrameCheckpointBridge)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${FROZEN_FRAME_BUILD}" ../../models/integrated)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden -fvisibility-inlines-hidden)
 target_link_libraries(${target} PRIVATE Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
endforeach()
add_executable(physics_checkpoint Driver.cpp)
target_compile_features(physics_checkpoint PRIVATE cxx_std_17)
target_compile_options(physics_checkpoint PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(physics_checkpoint PRIVATE CurveCheckpointBridge FrameCheckpointBridge Sofa.SimpleApi Sofa.Simulation.Graph Sofa.DefaultType)
