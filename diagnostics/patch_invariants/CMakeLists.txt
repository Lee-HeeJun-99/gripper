cmake_minimum_required(VERSION 3.22)
project(PatchInvariants LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Sofa.Component.StateContainer REQUIRED)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/PatchInvariants.cpp")
add_custom_command(OUTPUT "${generated}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${generated}" DEPENDS generate.py ../curved_contact/Invariants.cpp VERBATIM)
add_executable(patch_invariants "${generated}")
target_include_directories(patch_invariants PRIVATE "${PATCH_BUILD}" "${FROZEN_CURVE_BUILD}/material_template/birth_template/legacy" "${CMAKE_CURRENT_SOURCE_DIR}/../../models/integrated")
target_compile_features(patch_invariants PRIVATE cxx_std_17)
target_compile_options(patch_invariants PRIVATE -O3 -march=native -ffp-contract=off -Wall -Wextra -Wpedantic)
target_link_libraries(patch_invariants PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core Sofa.Component.StateContainer)
