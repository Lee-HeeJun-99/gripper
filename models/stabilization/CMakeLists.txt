cmake_minimum_required(VERSION 3.22)
project(ReGripStabilization LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../integrated/Components.cpp" integrated_source)
string(REPLACE "void registerObjects(sofa::core::ObjectFactory*f)" "void registerIntegratedObjects(sofa::core::ObjectFactory*f)" integrated_source "${integrated_source}")
string(REPLACE "const char*getModuleName()" "const char*getIntegratedModuleName()" integrated_source "${integrated_source}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Integrated.inc" CONTENT "${integrated_source}")
add_library(ReGripStabilization SHARED Audit.cpp)
target_include_directories(ReGripStabilization PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../integrated")
target_compile_features(ReGripStabilization PUBLIC cxx_std_17)
target_compile_options(ReGripStabilization PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ReGripStabilization PUBLIC Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Simulation.Core)
