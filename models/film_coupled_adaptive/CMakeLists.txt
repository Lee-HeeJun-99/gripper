cmake_minimum_required(VERSION 3.22)
project(ReGripCoupledAdaptive LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.LinearAlgebra REQUIRED)
find_package(Sofa.Simulation.Core REQUIRED)
find_package(Eigen3 REQUIRED)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../film_value_energy" "${CMAKE_CURRENT_BINARY_DIR}/value")
add_library(ReGripCoupledAdaptive SHARED AdaptiveOwner.cpp)
target_compile_features(ReGripCoupledAdaptive PRIVATE cxx_std_17)
target_compile_options(ReGripCoupledAdaptive PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
target_link_libraries(ReGripCoupledAdaptive PRIVATE ReGripFilmValue Eigen3::Eigen Sofa.Core Sofa.DefaultType Sofa.LinearAlgebra Sofa.Simulation.Core)
