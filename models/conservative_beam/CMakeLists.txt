cmake_minimum_required(VERSION 3.22)
project(ReGripConservativeBeam LANGUAGES CXX)
find_package(Sofa.Core REQUIRED)
find_package(Sofa.DefaultType REQUIRED)
find_package(Sofa.Helper REQUIRED)
find_package(Sofa.Component.StateContainer REQUIRED)
add_library(ReGripConservativeBeam SHARED ConservativeBeam.cpp)
add_executable(test_conservative_beam ../../tests/test_conservative_beam.cpp)
foreach(target ReGripConservativeBeam test_conservative_beam)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${target} PRIVATE Sofa.Core Sofa.DefaultType Sofa.Helper Sofa.Component.StateContainer)
endforeach()
enable_testing()
add_test(NAME conservative_beam COMMAND test_conservative_beam)
execute_process(COMMAND git show 4d19774:models/conservative_beam/EnergyKernel.h WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE reference RESULT_VARIABLE reference_status)
if(NOT reference_status EQUAL 0)
 message(FATAL_ERROR "Immutable dense reference commit unavailable")
endif()
string(REPLACE "namespace regrip_beam" "namespace beam_reference_v1" reference "${reference}")
string(REPLACE "#include \"../finite_rotation/Kernel.h\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/../finite_rotation/Kernel.h\"" reference "${reference}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/BeamReferenceV1.inc" CONTENT "${reference}")
add_executable(test_conservative_beam_equivalence ../../tests/test_conservative_beam_equivalence.cpp)
target_include_directories(test_conservative_beam_equivalence PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(test_conservative_beam_equivalence PRIVATE cxx_std_17)
target_compile_options(test_conservative_beam_equivalence PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME conservative_beam_dense_equivalence COMMAND test_conservative_beam_equivalence)
